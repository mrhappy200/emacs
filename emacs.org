  (defun my/popper-toggle-vterm ()
    "Show or hide a dedicated dropdown vterm and select it when shown."
    (interactive)
    (let* ((buffer-name "*vterm-dropdown*")
           (buffer (get-buffer buffer-name))
           (window (and buffer (get-buffer-window buffer))))
      (if window
          ;; This is specifically the terminal window, rather than merely
          ;; Popper's most recently opened popup.
          (delete-window window)
        (unless (buffer-live-p buffer)
          (let ((vterm-buffer-name buffer-name))
            (vterm)
            (setq buffer (current-buffer))))
        (let ((terminal-window (display-buffer buffer)))
          (select-window terminal-window)))))

  (defun my/org-skip-daily-recurring ()
    "Skip daily recurring and weekday-only scheduled entries."
    (let ((subtree-end (save-excursion (org-end-of-subtree t))))
      (when (save-excursion
              (or
               ;; Ordinary daily repeaters: +1d, ++1d, .+1d.
               (re-search-forward
                "<[^>\n]+\\(\\.\\+\\|\\+\\+\\|\\+\\)1d[^>]*>"
                subtree-end t)

               ;; Weekday-only diary-style scheduled timestamp:
               ;; SCHEDULED: <%%(memq (calendar-day-of-week date) '(1 2 3 4 5))>
               (re-search-forward
                (regexp-quote
                 "SCHEDULED: <%%(memq (calendar-day-of-week date) '(1 2 3 4 5))>")
                subtree-end t)))
        subtree-end)))
